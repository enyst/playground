Keep tablet sections
within reach

At 768–1023px, Settings and Customize pages gain a compact control above their content. Opening it reveals the existing sibling links without leaving the current page. Phone hubs and desktop sidebars retain their current behavior.

★ The product change is navigation presentation only: it reuses the existing link renderers and backend filtering. It adds no API dependency or durable state.

Start with src/components/shared/compact-section-navigation.tsx:74 and src/components/features/skills/extensions-navigation.tsx:176. Base 553d451911 · source head c0ed95678d.

The current page keeps its context

The compact layout introduced for tablets preserved page width but removed the secondary sidebar. This change keeps the hub available while adding a direct, discoverable section switcher. Both flows use two selections; the improvement is retaining context and exposing the choices on the page.

Sources: src/routes/mcp.tsx:116, src/components/features/skills/extensions-mobile-hub.tsx:1, src/components/shared/compact-section-navigation.tsx:21, src/routes/skills-settings.tsx:141.

A small disclosure owns only open state

CompactSectionNavigation checks the existing 767px and 1023px breakpoints. The Settings and Customize wrappers derive the current label from the navigation context. SectionDisclosure holds only a transient open flag. Route changes remount the disclosure; link activation and outside clicks close it. Escape closes it and returns focus to its 44px trigger.

Sources: src/components/shared/compact-section-navigation.tsx:21, src/components/shared/compact-section-navigation.tsx:74.

+  label: string;
+  currentLabel?: string;

+  children: React.ReactNode;

+function SectionDisclosure({
+  label,
+  currentLabel,
+  testId,
+  children,
+}: CompactSectionNavigationProps) {
+  const [isOpen, setIsOpen] = useState(false);
+  const menuId = useId();
+  const triggerRef = useRef<HTMLButtonElement>(null);
+  const containerRef = useClickOutsideElement<HTMLDivElement>(() =>
+    setIsOpen(false),
+  );
+  useCloseOnEscape(isOpen, () => setIsOpen(false), triggerRef);
+
+  return (
+    <div ref={containerRef} className="relative" data-testid={testId}>
+      <button
+        ref={triggerRef}
+        type="button"
+        aria-label={`${label}: ${currentLabel}`}
+        aria-expanded={isOpen}
+        aria-controls={menuId}
+        onClick={() => setIsOpen(!isOpen)}
+        className={cn(dropdownTriggerShellClassName, "min-h-11 text-sm")}
+      >
+        <span className="shrink-0 text-muted">{label}</span>
+        <span className="min-w-0 flex-1 truncate text-left text-contrast">
+          {currentLabel}
+        </span>
+        <ChevronDown className="size-4 shrink-0" aria-hidden />
+      </button>
+      {isOpen && (
+        // Link activation bubbles here for mouse and keyboard alike. Child
+        // anchors own navigation; this only dismisses their disclosure.
+        // eslint-disable-next-line jsx-a11y/no-noninteractive-element-interactions -- delegated link activation, not a standalone click target
+        <nav
+          id={menuId}
+          aria-label={label}
+          className="absolute inset-x-0 top-full z-50 mt-1 max-h-80 overflow-y-auto rounded-md border border-border-subtle bg-tertiary p-1 shadow-lg"
+          onClick={(event) => {
+            if (event.target instanceof Element && event.target.closest("a")) {
+              setIsOpen(false);
+            }
+          }}
+        >
+          {children}
+        </nav>
+      )}
+    </div>
+  );
+}
+
+/** Keep sibling pages reachable while the secondary sidebar is hidden. */
+export function CompactSectionNavigation(props: CompactSectionNavigationProps) {
+  const { currentPath } = useNavigation();
+  const isCompact = useBreakpoint(SETTINGS_COMPACT_MAX_WIDTH);
+  const isPhone = useBreakpoint(SIDEBAR_RAIL_COLLAPSE_MAX_WIDTH);
+
+  if (!isCompact || isPhone || !props.currentLabel) return null;
+
+  // Route changes also dismiss a disclosure opened before browser Back/Forward.
+  return <SectionDisclosure key={currentPath} {...props} />;
